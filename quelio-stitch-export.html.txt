<!-- JavaScript for live UI interactions -->
<script data-purpose="event-handlers">
    // Live word count calculation
    const contentInput = document.getElementById('quiz-content-input');
    const wordCountDisplay = document.getElementById('word-count-display');
    const clearBtn = document.getElementById('btn-clear');
    const studyToggle = document.getElementById('study-mode-toggle');
    const studyLabel = document.getElementById('study-mode-label');

    if (contentInput && wordCountDisplay) {
      contentInput.addEventListener('input', () => {
        const text = contentInput.value.trim();
        const count = text ? text.split(/\s+/).filter(Boolean).length : 0;
        wordCountDisplay.textContent = count;
      });
    }

    if (clearBtn && contentInput && wordCountDisplay) {
      clearBtn.addEventListener('click', () => {
        contentInput.value = '';
        wordCountDisplay.textContent = '0';
        contentInput.focus();
      });
    }

    if (studyToggle && studyLabel) {
      studyToggle.addEventListener('change', (e) => {
        studyLabel.textContent = e.target.checked ? 'On' : 'Off';
      });
    }
  </script>
